/**
 * Gestionale_Clienti — foglio di lavoro precompilato (v1).
 *
 * Aggiunge alla scheda Clienti un link per riga: lo clicchi e ti si apre il
 * rapportino di intervento già compilato con i dati di quel cliente, pronto
 * da stampare. Il prezzo NON viene compilato: resta vuoto di proposito.
 *
 * Il layout NON sta qui dentro: viene scaricato dal sito a ogni apertura.
 * La fonte unica è il modulo cartaceo `moduli/rapportino-intervento.html`;
 * `node moduli/genera-template-appsscript.mjs` ne ricava il modello coi
 * segnaposto e lo mette in `sito/public/moduli/`, che va online col deploy.
 * Quindi: cambi il foglio di carta, rigeneri, fai il deploy — e il
 * precompilato è già allineato senza toccare Apps Script.
 *
 * ── Installazione ─────────────────────────────────────────────────────────
 * Va nel progetto Apps Script LEGATO al foglio (Estensioni → Apps Script),
 * accanto a Codice.gs che contiene l'automazione lead → cliente. I menu
 * esistono solo negli script legati a un documento.
 * Basta incollare e salvare: il menu compare alla riapertura del foglio.
 *
 * Niente app web: su questo account non vengono servite (qualsiasi
 * distribuzione risponde "Impossibile aprire il file" senza mai far partire
 * lo script). Il `doGet` qui sotto resta solo come avanzo innocuo.
 *
 * Copia di riserva di questo script: cartella acquadirete/documenti.
 */

// Il modello del rapportino, pubblicato col sito.
const URL_MODELLO = 'https://www.acquadirete.it/moduli/rapportino-modello.html';

// Il foglio Gestionale_Clienti, aperto per ID.
// Questo script vive in un progetto AUTONOMO, non legato al foglio: i progetti
// legati a un foglio, su questo account, non riescono a servire app web
// (verificato il 7 ago 2026 — un progetto autonomo con lo stesso codice
// funziona, quello legato dà sempre "Impossibile aprire il file", su qualsiasi
// distribuzione e senza mai registrare un'esecuzione).
const ID_FOGLIO = '1jmR0DXP_25ZTniBgemRj0hEmXxcrCdxj89y4DxciHBk';

function foglio() {
  // Legato al foglio: getActive() basta. openById resta come rete di sicurezza
  // se un domani lo script girasse da un progetto autonomo.
  return SpreadsheetApp.getActive() || SpreadsheetApp.openById(ID_FOGLIO);
}

// Dal 7 ago 2026 anagrafica e impianto stanno su una scheda sola: una riga
// per cliente. Prima erano "Clienti" e "Impianti-Manutenzioni" separate.
const SCHEDA_CLIENTI = 'Clienti-Impianti';

// Le colonne si cercano PER NOME dell'intestazione, non per posizione.
// Tre volte in due giorni le colonne del Gestionale si sono spostate e lo
// script ha continuato a leggere gli stessi indici, riempiendo i campi con i
// dati sbagliati senza dare errore: il modo peggiore di rompersi, perché il
// foglio esce comunque e sembra a posto. Cercando l'intestazione, spostare o
// inserire colonne non fa più danni; e se un'intestazione sparisce o viene
// rinominata, lo script si ferma dicendo quale, invece di indovinare.
//
// Per ogni campo bastano una parola o due che compaiano nell'intestazione,
// confrontate senza accenti, maiuscole né punteggiatura.
const INTESTAZIONI = {
  CODICE:        ['codice'],
  NOME:          ['nome', 'ragione sociale'],
  INDIRIZZO:     ['indirizzo'],
  CITTA:         ['citta', 'comune'],
  CONTATTO:      ['telefono', 'contatto'],
  TIPO:          ['tipo impianto', 'impianto'],
  INSTALLAZIONE: ['data installazione', 'installazione']
};

/** Toglie accenti, maiuscole, punteggiatura e spazi doppi: così "Città" e
 *  "CITTA'" combaciano, e "Telefono / Contatto" combacia con "telefono". */
function normalizza(s) {
  return String(s == null ? '' : s)
    .toLowerCase()
    .replace(/[àáâä]/g, 'a').replace(/[èéêë]/g, 'e').replace(/[ìíîï]/g, 'i')
    .replace(/[òóôö]/g, 'o').replace(/[ùúûü]/g, 'u')
    .replace(/[^a-z0-9 ]+/g, ' ')
    .replace(/\s+/g, ' ')
    .trim();
}

/** Trova la riga delle intestazioni (quella che contiene "codice") e mappa
 *  ogni campo alla sua colonna. Restituisce { riga, col: {CODICE: n, ...} }. */
function mappaColonne(sh) {
  const quante = Math.min(sh.getLastRow(), 15);
  const griglia = sh.getRange(1, 1, quante, sh.getLastColumn()).getValues();

  for (let r = 0; r < griglia.length; r++) {
    const celle = griglia[r].map(normalizza);
    if (celle.indexOf('codice') === -1) continue; // non è la riga dei titoli

    const col = {};
    const mancanti = [];
    Object.keys(INTESTAZIONI).forEach(function (campo) {
      const chiavi = INTESTAZIONI[campo];
      let trovata = -1;
      // Prima le corrispondenze esatte, poi quelle parziali: così "Codice"
      // non viene rubato da "Codice destinatario" se un giorno esistesse.
      for (let k = 0; k < chiavi.length && trovata === -1; k++) trovata = celle.indexOf(chiavi[k]);
      for (let k = 0; k < chiavi.length && trovata === -1; k++) {
        for (let c = 0; c < celle.length && trovata === -1; c++) {
          if (celle[c] && celle[c].indexOf(chiavi[k]) !== -1) trovata = c;
        }
      }
      if (trovata === -1) mancanti.push(chiavi[0]);
      else col[campo] = trovata + 1;
    });

    if (mancanti.length) {
      throw new Error('Nella scheda "' + sh.getName() + '" non trovo le colonne: ' +
        mancanti.join(', ') + '. Se le hai rinominate, aggiorna INTESTAZIONI nello script.');
    }
    return { riga: r + 1, col: col };
  }
  throw new Error('Nella scheda "' + sh.getName() + '" non trovo la riga delle intestazioni (nessuna cella "Codice" nelle prime ' + quante + ' righe).');
}


/** Voce di menu nel foglio. Trigger semplice: compare da sola alla riapertura. */
function onOpen() {
  SpreadsheetApp.getUi()
    .createMenu('Foglio di lavoro')
    .addItem('Stampa per questo cliente', 'stampaRapportino')
    .addToUi();
}

/** Genera il rapportino del cliente sulla riga in cui hai il cursore. */
function stampaRapportino() {
  const ui = SpreadsheetApp.getUi();
  const sh = SpreadsheetApp.getActive().getActiveSheet();

  if (sh.getName() !== SCHEDA_CLIENTI) {
    ui.alert('Mettiti sulla scheda "' + SCHEDA_CLIENTI + '", sulla riga del cliente.');
    return;
  }
  const riga = sh.getActiveRange().getRow();

  let out, codice;
  try {
    const mappa = mappaColonne(sh);
    codice = String(sh.getRange(riga, mappa.col.CODICE).getValue() || '').trim().toUpperCase();
    if (!/^C\d+$/.test(codice)) {
      ui.alert('Riga ' + riga + ': nella colonna "Codice" non c\'è un codice cliente (tipo C012).');
      return;
    }
    out = costruisci(codice);
  } catch (err) {
    ui.alert(String(err.message || err));
    return;
  }
  ui.showModalDialog(
    HtmlService.createHtmlOutput(lanciatore(out.chi, out.modulo)).setWidth(420).setHeight(230),
    'Foglio di lavoro'
  );
}


/** Resta per compatibilità con i link già scritti in colonna I. */
function doGet(e) {
  const codice = String((e && e.parameter && e.parameter.c) || '').trim().toUpperCase();
  if (!codice) return errore('Manca il codice cliente nel link (serve ?c=C0xx).');
  let out;
  try {
    out = costruisci(codice);
  } catch (err) {
    return errore(String(err.message || err));
  }
  return HtmlService.createHtmlOutput(lanciatore(out.chi, out.modulo));
}


/** Il cuore: dal codice cliente al modulo compilato, pronto da mostrare.
 *  Usato sia dal menu che dal doGet. */
function costruisci(codice) {
  const ss = foglio();
  const sh = ss.getSheetByName(SCHEDA_CLIENTI);
  if (!sh) throw new Error('Non trovo la scheda "' + SCHEDA_CLIENTI + '".');

  const mappa = mappaColonne(sh);
  const COL = mappa.col;
  // Una riga sola per cliente: anagrafica e impianto insieme.
  const riga = trovaPerCodice(sh, codice, COL.CODICE);
  if (!riga) throw new Error('Nessun cliente col codice ' + codice + ' nella scheda ' + SCHEDA_CLIENTI + '.');

  const nome = String(riga[COL.NOME - 1] || '').trim();
  const contatto = String(riga[COL.CONTATTO - 1] || '').trim();
  const tipo = String(riga[COL.TIPO - 1] || '').trim();
  const caselle = caselleImpianto(tipo);
  const luogo = dividiCitta(String(riga[COL.CITTA - 1] || ''));

  const valori = {
    CODICE: codice,
    NOME: nome,
    INDIRIZZO: String(riga[COL.INDIRIZZO - 1] || '').trim(),
    CAP: luogo.cap,
    COMUNE: luogo.comune,
    TELEFONO: soloTelefono(contatto),
    EMAIL: soloEmail(contatto),
    DATA_INSTALLAZIONE: dataItaliana(riga[COL.INSTALLAZIONE - 1]),
    // In Impianti-Manutenzioni "Tipo impianto" contiene il modello commerciale
    // ("Nathalie 100 GPD", "Chloé con gassata"), quindi va sempre nel campo
    // Modello. Le caselle si barrano solo quando il testo lo dice chiaramente.
    MODELLO: tipo,
    K_OSMOSI: caselle.osmosi,
    K_MICRO: caselle.micro,
    K_CARBONI: caselle.carboni,
    K_GASATORE: caselle.gasatore
  };

  let modulo = scaricaModello();
  Object.keys(valori).forEach(function (chiave) {
    modulo = modulo.split('{{' + chiave + '}}').join(fugaHtml(valori[chiave]));
  });

  // Stampa automatica appena la pagina è pronta.
  modulo = modulo.replace('</body>',
    '<script>window.onload=function(){setTimeout(function(){window.print();},400);};<\/script></body>');

  return { chi: nome || codice, modulo: modulo };
}

/** Scarica il modello dal sito, tenendolo in cache un'ora per non rifare la
 *  chiamata a ogni clic. Se il sito non risponde si ferma con un messaggio
 *  chiaro invece di stampare un foglio mezzo vuoto. */
function scaricaModello() {
  const cache = CacheService.getScriptCache();
  const salvato = cache.get('modello');
  if (salvato) return salvato;

  const res = UrlFetchApp.fetch(URL_MODELLO, { muteHttpExceptions: true });
  if (res.getResponseCode() !== 200) {
    throw new Error('Modello non raggiungibile (HTTP ' + res.getResponseCode() + '): ' + URL_MODELLO);
  }
  const testo = res.getContentText('UTF-8');
  cache.put('modello', testo, 3600);
  return testo;
}

/** Svuota la cache del modello: da lanciare dopo un deploy del sito se vuoi
 *  vedere subito le modifiche senza aspettare l'ora. */
function aggiornaModello() {
  CacheService.getScriptCache().remove('modello');
  Logger.log('Modello ricaricato dal sito al prossimo clic.');
}

/** Cerca il codice nella colonna indicata, senza dare per scontata la riga
 *  d'inizio: così funziona qualunque sia l'altezza dell'intestazione. */
function trovaPerCodice(sh, codice, colCodice) {
  if (!sh) return null;
  const ultima = sh.getLastRow();
  if (ultima < 1) return null;
  const righe = sh.getRange(1, 1, ultima, sh.getLastColumn()).getValues();
  for (let i = 0; i < righe.length; i++) {
    if (String(righe[i][colCodice - 1] || '').trim().toUpperCase() === codice) return righe[i];
  }
  return null;
}

/** Dal "tipo impianto" ricava quali caselle barrare. Nel dubbio non barra:
 *  una casella sbagliata è peggio di una casella vuota, perché il tecnico
 *  legge il foglio di fretta e si fida. "GPD" (gallons per day) compare solo
 *  sugli osmosi; "soda/gassata/CO2/bombola" solo dove c'è il gasatore. */
function caselleImpianto(tipo) {
  const t = String(tipo).toLowerCase();
  const r = {
    osmosi:   /(osmosi|\bgpd\b)/.test(t) ? 'x' : '',
    micro:    /microfiltr/.test(t) ? 'x' : '',
    carboni:  /carbon/.test(t) ? 'x' : '',
    gasatore: /(gasat|frizzant|gassat|soda|co2|bombola)/.test(t) ? 'x' : ''
  };
  r.trovata = Boolean(r.osmosi || r.micro || r.carboni || r.gasatore);
  return r;
}

/** In Clienti la città spesso contiene anche il CAP ("50025 MONTESPERTOLI"),
 *  ma non sempre ("FIRENZE"): quando c'è lo separa, altrimenti lascia il CAP
 *  vuoto da riempire a penna. */
function dividiCitta(citta) {
  const s = String(citta).replace(/\s+/g, ' ').trim();
  const m = s.match(/^(\d{5})\s+(.*)$/);
  return m ? { cap: m[1], comune: m[2].trim() } : { cap: '', comune: s };
}

/** La colonna Contatto tiene a volte il telefono, a volte l'email, a volte
 *  entrambi: li separa così finiscono nei campi giusti del modulo. */
function soloTelefono(contatto) {
  const pezzi = String(contatto).split(/[\s;,\/]+/);
  for (let i = 0; i < pezzi.length; i++) {
    if (pezzi[i].indexOf('@') === -1 && (pezzi[i].match(/\d/g) || []).length >= 6) return pezzi[i];
  }
  return contatto.indexOf('@') === -1 ? contatto : '';
}

function soloEmail(contatto) {
  const pezzi = String(contatto).split(/[\s;,\/]+/);
  for (let i = 0; i < pezzi.length; i++) {
    if (pezzi[i].indexOf('@') !== -1) return pezzi[i];
  }
  return '';
}

function dataItaliana(v) {
  if (!v) return '';
  if (Object.prototype.toString.call(v) === '[object Date]') {
    return Utilities.formatDate(v, Session.getScriptTimeZone(), 'dd/MM/yyyy');
  }
  return String(v).trim();
}

/** Controlla che le colonne di Clienti-Impianti siano ancora dove
 *  l'automazione lead -> cliente (Codice.gs) si aspetta di trovarle.
 *  Quella scrive per posizione, quindi se le colonne si spostano
 *  riempirebbe i campi sbagliati senza accorgersene: meglio fermarla.
 *  Restituisce '' se va tutto bene, altrimenti il messaggio da mostrare. */
function verificaLayout(sh) {
  const attese = [
    [1, 'codice'], [2, 'nome'], [3, 'indirizzo'], [4, 'citta'], [5, 'provincia'],
    [6, 'telefono'], [7, 'referente'], [8, 'zona giro'], [9, 'tipo impianto'],
    [12, 'prossima manutenzione'], [13, 'giorni'], [14, 'stato'], [15, 'note']
  ];

  const quante = Math.min(sh.getLastRow(), 15);
  const griglia = sh.getRange(1, 1, quante, sh.getLastColumn()).getValues();
  let titoli = null;
  for (let r = 0; r < griglia.length && !titoli; r++) {
    const celle = griglia[r].map(normalizza);
    if (celle.indexOf('codice') !== -1) titoli = celle;
  }
  if (!titoli) return 'Non trovo la riga delle intestazioni in ' + sh.getName() + '.';

  const fuori = [];
  attese.forEach(function (a) {
    const t = titoli[a[0] - 1] || '';
    if (t.indexOf(a[1]) === -1) fuori.push(a[0] + ': aspetto "' + a[1] + '", trovo "' + (t || 'vuota') + '"');
  });
  if (!fuori.length) return '';
  return 'Le colonne di ' + sh.getName() + ' si sono spostate: non copio il lead per non scrivere nei campi sbagliati. ' + fuori.join('; ') + '.';
}

function fugaHtml(s) {
  return String(s == null ? '' : s)
    .replace(/&/g, '&amp;').replace(/</g, '&lt;').replace(/>/g, '&gt;').replace(/"/g, '&quot;');
}

/** Messaggio leggibile al posto di una schermata bianca quando il link è
 *  sbagliato o il cliente non esiste. */
function errore(messaggio) {
  return HtmlService.createHtmlOutput(
    '<div style="font-family:Arial,Helvetica,sans-serif;color:#01278B;padding:20px;text-align:center">' +
    '<p><b>Foglio di lavoro</b></p><p>' + fugaHtml(messaggio) + '</p></div>'
  ).setTitle('Foglio di lavoro');
}

/** L'app web di Google serve le pagine dentro un iframe, dove le regole di
 *  stampa @page (A4, margini) non vengono applicate. Per avere in stampa
 *  esattamente il modulo che abbiamo verificato, il modulo viene aperto in una
 *  scheda vera come blob: lì il CSS di stampa vale per intero. */
function lanciatore(chi, modulo) {
  const dati = JSON.stringify(modulo).replace(/<\//g, '<\\/');
  const html =
    '<!DOCTYPE html><html><head><meta charset="utf-8"><base target="_blank">' +
    '<style>body{font-family:Arial,Helvetica,sans-serif;color:#01278B;padding:18px;text-align:center}' +
    'button{font-size:15px;padding:10px 22px;border:0;border-radius:6px;background:#01278B;color:#fff;cursor:pointer}' +
    'p{font-size:13px}small{color:#4a68ad}</style></head><body>' +
    '<p>Foglio di lavoro di <b>' + fugaHtml(chi) + '</b>.</p>' +
    '<button onclick="apri()">Apri e stampa</button>' +
    '<p><small id="avviso"></small></p>' +
    '<script>var M=' + dati + ';' +
    'function apri(){' +
    ' var u=URL.createObjectURL(new Blob([M],{type:"text/html"}));' +
    ' var w=window.open(u,"_blank");' +
    ' if(!w){document.getElementById("avviso").textContent=' +
    '   "Il browser ha bloccato la finestra: consenti i popup per questo sito e riprova.";}' +
    '}<\/script></body></html>';
  return html; // stringa: chi chiama decide se farne una finestra o una pagina
}
